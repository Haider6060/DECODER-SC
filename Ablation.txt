# 1. Load dataset
seu <- readRDS("GSE185204_lung cancer_dataset.rds")

set.seed(123)
cells_use <- sample(colnames(seu), 20000)
sim_seu <- subset(seu, cells = cells_use)

DefaultAssay(sim_seu) <- "RNA"
sim_seu <- JoinLayers(sim_seu)

sim_seu <- NormalizeData(sim_seu, verbose = FALSE)
sim_seu <- FindVariableFeatures(sim_seu, verbose = FALSE)
sim_seu <- ScaleData(sim_seu, verbose = FALSE)
sim_seu <- RunPCA(sim_seu, verbose = FALSE)
sim_seu <- FindNeighbors(sim_seu, dims = 1:20, verbose = FALSE)
sim_seu <- FindClusters(sim_seu, resolution = 0.5, verbose = FALSE)

Idents(sim_seu) <- sim_seu$seurat_clusters
avg_expr <- AverageExpression(sim_seu, layer = "scale.data")[[1]]

# 2. DoRothEA regulons
data("dorothea_hs", package = "dorothea")
regulons_df <- dorothea_hs[dorothea_hs$confidence %in% c("A", "B", "C"), ]
regulons <- split(regulons_df$target, regulons_df$tf)

# 3. AUCell
rankings <- AUCell_buildRankings(avg_expr, plotStats = FALSE)
tf_activity <- AUCell_calcAUC(regulons, rankings)
tf_activity_mat <- as.matrix(getAUC(tf_activity))

common_genes <- intersect(rownames(avg_expr), regulons_df$target)
state_expr <- avg_expr[common_genes, , drop = FALSE]
rules <- regulons_df[regulons_df$target %in% common_genes, ]
rules$gene <- rules$target

# 4. Simulated TFs
simulated_tfs <- c("PPARA", "BACH1", "RUNX3", "MAFB", "TBX21")

# ------------------------------------------------------------
# 5. Create perturbed object (ORIGINAL METHOD)
# ------------------------------------------------------------

set.seed(123)

sim_seu_perturbed <- sim_seu

sim_targets <- rules$target[rules$tf %in% simulated_tfs]
sim_targets <- unique(sim_targets)

expr_mat <- GetAssayData(sim_seu_perturbed, assay = "RNA", layer = "data")

cells_perturb <- sample(colnames(sim_seu_perturbed), 1200)

for (g in sim_targets) {
  if (g %in% rownames(expr_mat)) {
    expr_mat[g, cells_perturb] <- expr_mat[g, cells_perturb] * 0.1
  }
}

sim_seu_perturbed <- SetAssayData(
  sim_seu_perturbed,
  assay = "RNA",
  layer = "data",
  new.data = expr_mat
)

# ------------------------------------------------------------
# 6. Full DECODER-SC function
# ------------------------------------------------------------

run_decoder_full <- function(obj) {
  
  Idents(obj) <- obj$seurat_clusters
  avg_expr <- AverageExpression(obj, layer = "scale.data")[[1]]
  rankings <- AUCell_buildRankings(avg_expr, plotStats = FALSE)
  tf_activity <- AUCell_calcAUC(regulons, rankings)
  tf_activity_mat <- as.matrix(getAUC(tf_activity))
  common_genes <- intersect(rownames(avg_expr), regulons_df$target)
  state_expr <- avg_expr[common_genes, , drop = FALSE]
  rules <- regulons_df[regulons_df$target %in% common_genes, ]
  
  q <- 0.75
  tf_state <- tf_activity_mat > apply(tf_activity_mat, 1, quantile, probs = q)
  gene_state <- state_expr > apply(state_expr, 1, quantile, probs = q)
  gene_z <- t(scale(t(state_expr)))
  
  discordance_mat <- matrix(0, nrow = nrow(rules), ncol = ncol(state_expr))
  
  for (i in seq_len(nrow(rules))) {
    tf  <- rules$tf[i]
    tg  <- rules$target[i]
    mor <- rules$mor[i]
    
    if (tf %in% rownames(tf_state)) {
      tf_on   <- tf_state[tf, ]
      gene_on <- gene_state[tg, ]
      
      if (mor == 1) {
        observed <- ifelse(tf_on & !gene_on, abs(gene_z[tg, ]), 0)
        expected <- mean(tf_on) * mean(!gene_on)
        discordance_mat[i, ] <- pmax(observed - expected, 0)
      }
      if (mor == -1) {
        observed <- ifelse(tf_on & gene_on, abs(gene_z[tg, ]), 0)
        expected <- mean(tf_on) * mean(gene_on)
        discordance_mat[i, ] <- pmax(observed - expected, 0)
      }
    }
  }
  
  rules$discordance_score <- rowMeans(discordance_mat)
  tf_summary <- aggregate(discordance_score ~ tf, rules, mean)
  tf_summary <- tf_summary[order(-tf_summary$discordance_score), ]
  return(tf_summary)
}

perturbed_tf <- run_decoder_full(sim_seu_perturbed)

# 7. No-sign ablation function
# ------------------------------------------------------------

run_decoder_no_sign <- function(obj) {
  
  if (!"percent.mt" %in% colnames(obj@meta.data)) {
    obj[["percent.mt"]] <- PercentageFeatureSet(obj, "^MT-")
  }
  
  obj <- subset(obj, subset = nFeature_RNA > 200 & percent.mt < 20)
  obj <- NormalizeData(obj, verbose = FALSE)
  obj <- FindVariableFeatures(obj, verbose = FALSE)
  obj <- ScaleData(obj, verbose = FALSE)
  obj <- RunPCA(obj, verbose = FALSE)
  obj <- FindNeighbors(obj, dims = 1:20, verbose = FALSE)
  obj <- FindClusters(obj, resolution = 0.5, verbose = FALSE)
  
  Idents(obj) <- obj$seurat_clusters
  avg_expr <- AverageExpression(obj, layer = "scale.data")[[1]]
  rankings <- AUCell_buildRankings(avg_expr, plotStats = FALSE)
  tf_activity <- AUCell_calcAUC(regulons, rankings)
  tf_activity_mat <- as.matrix(getAUC(tf_activity))
  common_genes <- intersect(rownames(avg_expr), regulons_df$target)
  state_expr <- avg_expr[common_genes, , drop = FALSE]
  rules <- regulons_df[regulons_df$target %in% common_genes, ]
  
  q <- 0.75
  tf_state <- tf_activity_mat > apply(tf_activity_mat, 1, quantile, probs = q)
  gene_state <- state_expr > apply(state_expr, 1, quantile, probs = q)
  gene_z <- t(scale(t(state_expr)))
  
  discordance_mat <- matrix(0, nrow = nrow(rules), ncol = ncol(state_expr))
  
  for (i in seq_len(nrow(rules))) {
    tf <- rules$tf[i]
    tg <- rules$target[i]
    
    if (tf %in% rownames(tf_state)) {
      tf_on <- tf_state[tf, ]
      gene_on <- gene_state[tg, ]
      
      observed <- ifelse(tf_on & !gene_on, abs(gene_z[tg, ]), 0)
      expected <- mean(tf_on) * mean(!gene_on)
      discordance_mat[i, ] <- pmax(observed - expected, 0)
    }
  }
  
  rules$discordance_score <- rowMeans(discordance_mat)
  tf_summary <- aggregate(discordance_score ~ tf, rules, mean)
  tf_summary <- tf_summary[order(-tf_summary$discordance_score), ]
  return(tf_summary)
}
